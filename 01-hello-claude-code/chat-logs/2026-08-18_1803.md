## Wrote src/main.py (in two guided chunks)

```python
import os
import anthropic
from dotenv import load_dotenv

load_dotenv()

client = anthropic.Anthropic()

response = client.messages.create(
    model="claude-haiku-4-5",
    max_tokens=100,
    messages=[
        {"role": "user", "content": "Say hello and tell me one fact about Python in one sentence."}
    ]
)

print(response.content[0].text)
```
